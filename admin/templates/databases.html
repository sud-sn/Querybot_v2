Set up a read-only service user in Snowflake

Your Snowflake administrator runs this once. Replace the names in angle brackets. The public key comes from Generate a key pair above, before you save, or from your own key.

CREATE ROLE IF NOT EXISTS QUERYBOT_READER;
GRANT USAGE ON WAREHOUSE <WAREHOUSE> TO ROLE QUERYBOT_READER;
GRANT USAGE ON DATABASE <DATABASE> TO ROLE QUERYBOT_READER;
GRANT USAGE ON SCHEMA <DATABASE>.<SCHEMA> TO ROLE QUERYBOT_READER;
GRANT SELECT ON ALL TABLES IN SCHEMA <DATABASE>.<SCHEMA> TO ROLE QUERYBOT_READER;
GRANT SELECT ON ALL VIEWS IN SCHEMA <DATABASE>.<SCHEMA> TO ROLE QUERYBOT_READER;
GRANT SELECT ON FUTURE TABLES IN SCHEMA <DATABASE>.<SCHEMA> TO ROLE QUERYBOT_READER;
GRANT SELECT ON FUTURE VIEWS IN SCHEMA <DATABASE>.<SCHEMA> TO ROLE QUERYBOT_READER;

CREATE USER IF NOT EXISTS QUERYBOT_SVC
  TYPE = SERVICE
  DEFAULT_ROLE = QUERYBOT_READER
  DEFAULT_WAREHOUSE = <WAREHOUSE>
  RSA_PUBLIC_KEY = '<public key from QueryBot>';
GRANT ROLE QUERYBOT_READER TO USER QUERYBOT_SVC;

-- Only if QueryBot keeps its logs in Snowflake (Log export below): it creates
-- the log schema and its tables itself, and owns them. Nothing else is written.
GRANT CREATE SCHEMA ON DATABASE <DATABASE> TO ROLE QUERYBOT_READER;
